#include "Expediente_Medico.h"
#include "Utilidades.h"     // unir()

// Constructor: guarda el id y deja el expediente en blanco.
Expediente_Medico::Expediente_Medico(const std::string& idExpediente)
    : idExpediente(idExpediente)
{
    crearExpediente();      // seguro: no es virtual
}

// crearExpediente(): vacia las listas. El UML lo declara 'void' y sin parametros, asi que solo puede PREPARAR el objeto existente.
// Ademas sirve para "reiniciar" un expediente mas adelante.
void Expediente_Medico::crearExpediente() {
    alergias.clear();
    discapacidades.clear();
}

// [SOBRECARGA 1] Solo el nombre de la alergia.
void Expediente_Medico::agregarAlergia(const std::string& alergia) {
    alergias.push_back(alergia);          // push_back agrega al final del vector
}

// [SOBRECARGA 2] Nombre + severidad. REUTILIZA la primera version:
// arma "Polen (alta)" y se lo pasa a la version de un parametro.
void Expediente_Medico::agregarAlergia(const std::string& alergia,
                                       const std::string& severidad) {
    agregarAlergia(alergia + " (" + severidad + ")");
}

void Expediente_Medico::agregarDiscapacidad(const std::string& discapacidad) {
    discapacidades.push_back(discapacidad);
}

// "!" = NO, "||" = O.  Se lee: "alergias NO esta vacio  O  discapacidades NO esta vacio".
// Si la primera mitad es verdadera, C++ ni evalua la segunda (cortocircuito).
bool Expediente_Medico::tieneCondiciones() const {
    return !alergias.empty() || !discapacidades.empty();
}

std::string Expediente_Medico::getIdExpediente() const { return idExpediente; }
const std::vector<std::string>& Expediente_Medico::getAlergias() const { return alergias; }
const std::vector<std::string>& Expediente_Medico::getDiscapacidades() const { return discapacidades; }

// Devuelve UN vector nuevo con alergias y discapacidades juntas (una copia:
// quien lo reciba no puede danar los datos internos).
std::vector<std::string> Expediente_Medico::consultarCondiciones() const {
    std::vector<std::string> condiciones;
    for (const std::string& a : alergias) {          // "para cada a dentro de alergias"
        condiciones.push_back(a);
    }
    for (const std::string& d : discapacidades) {
        condiciones.push_back(d);
    }
    return condiciones;
}

// Version LEGIBLE para pantalla. Las hijas la llaman con
// Expediente_Medico::generarResumen() y le agregan sus datos propios.
std::string Expediente_Medico::generarResumen() const {
    // Operador ternario:  condicion ? valor_si_true : valor_si_false
    // std::string("ninguna") fuerza que ambos lados sean del mismo tipo.
    std::string textoAlergias =
        alergias.empty() ? std::string("ninguna") : unir(alergias, ',');
    std::string textoDiscapacidades =
        discapacidades.empty() ? std::string("ninguna") : unir(discapacidades, ',');

    // getCategoria() es VIRTUAL: devuelve "Lactante" o "Preescolar" segun la hija real.
    return "Expediente " + idExpediente + " [" + getCategoria() + "]" +
           " | Alergias: " + textoAlergias +
           " | Discapacidades: " + textoDiscapacidades;
}

// Version de TEXTO (formato de intercambio). Formato:  categoria|id|alergias(;)|discapacidades(;)
// Las hijas la llaman primero y luego agregan "|dato|dato...".
void Expediente_Medico::escribir(std::ostream& os) const {
    os << getCategoria() << '|' << idExpediente << '|'
       << unir(alergias, ';') << '|' << unir(discapacidades, ';');
}

// [OPERADOR <<] Delega en el metodo virtual escribir() (los operadores no
// pueden ser virtuales). Gracias a eso, "cout << *expediente" escribe
// los datos de la hija correcta aunque solo conozcamos al padre.
std::ostream& operator<<(std::ostream& os, const Expediente_Medico& expediente) {
    expediente.escribir(os);
    return os;
}

#pragma once // #pragma once evita que el header se incluya dos veces y cause errores de redefinicion.

#include <string>
#include <vector>
#include <ostream>

class Expediente_Medico {
private:
    std::string idExpediente;                 // identidad del expediente
    std::vector<std::string> alergias;        // vector: lista que crece sola
    std::vector<std::string> discapacidades;

public:
    Expediente_Medico(const std::string& idExpediente);

    // DESTRUCTOR VIRTUAL: obligatorio como buena practica en toda clase pensada
    // para herencia. Garantiza que al destruir un objeto mediante un puntero al
    // padre se destruya el objeto COMPLETO (la parte hija incluida).
    virtual ~Expediente_Medico() = default;

    // No es virtual a proposito: el constructor la llama, y un constructor del
    // padre NUNCA ejecuta versiones sobrescritas de las hijas (la parte hija
    // todavia no existe cuando se construye el padre).
    void crearExpediente();                               // deja el expediente en blanco
    std::vector<std::string> consultarCondiciones() const; // alergias + discapacidades juntas

    // [SOBRECARGA] Mismo nombre, distintos parametros. El compilador elige
    // cual usar mirando los argumentos (decision en tiempo de compilacion).
    void agregarAlergia(const std::string& alergia);
    void agregarAlergia(const std::string& alergia, const std::string& severidad);

    void agregarDiscapacidad(const std::string& discapacidad);
    bool tieneCondiciones() const;                        // true si hay al menos una
    std::string getIdExpediente() const;
    const std::vector<std::string>& getAlergias() const;           // devuelve sin copiar y sin permitir modificar
    const std::vector<std::string>& getDiscapacidades() const;

    // [POLIMORFISMO] Metodos que las hijas sobrescriben.
    // El enlace es DINAMICO: C++ elige la version mirando el tipo REAL del objeto
    // en tiempo de ejecucion, no el tipo del puntero.
    virtual std::string getCategoria() const = 0;     // "= 0": virtual PURO. Cada hija DEBE implementarlo.
    virtual std::string generarResumen() const;       // texto legible; las hijas lo extienden
    virtual void escribir(std::ostream& os) const;    // formato de texto; las hijas lo extienden
};

// [OPERADOR DE FLUJO] Un operador NO puede ser virtual, asi que este se delega
// en el metodo virtual escribir(). Asi "cout << expediente" funciona
// polimorficamente: cada hija escribe sus propios datos.
std::ostream& operator<<(std::ostream& os, const Expediente_Medico& expediente);

Ficha_Medica::Ficha_Medica(const std::string &idProceso,
                           const std::string &fecha,
                           const std::shared_ptr<Infante> &infante,
                           bool tieneGolpes,
                           float temperatura)
    : Registro(idProceso, fecha, infante), // 1) se construye la parte del padre
      tieneGolpes(tieneGolpes),            // 2) atributos propios
      temperatura(temperatura),
      estadoSaludAdecuado(false), // aun no se ha evaluado
      severidad(SeveridadIrregularidad::Ninguna),   // al inicio: sin irregularidades
      descripcionIrregularidad("")
{
}

// [SOBRECARGA 1] Usa el limite maximo por defecto (TEMP_NORMAL_MAX = 37.5).
// Delega en la version con parametro.
bool Ficha_Medica::evaluacionSalud() const
{
    return evaluacionSalud(TEMP_NORMAL_MAX);
}

// [SOBRECARGA 2] Esta sano si NO tiene golpes, la temperatura esta DENTRO del rango
// normal (de TEMP_NORMAL_MIN hasta el limite maximo, ambos incluidos) y no hay
// irregularidad grave. "!" = NO,  "&&" = Y.
bool Ficha_Medica::evaluacionSalud(float limiteTemperatura) const
{
   return !tieneGolpes
           && temperatura >= TEMP_NORMAL_MIN && temperatura <= limiteTemperatura
           && severidad != SeveridadIrregularidad::Grave;
}

// Registra una irregularidad: que es y que tan seria es.
bool Ficha_Medica::registrarIrregularidad(const std::string& descripcion,
                                          SeveridadIrregularidad nuevaSeveridad) {
    if (descripcion.empty() || nuevaSeveridad == SeveridadIrregularidad::Ninguna) {
        return false;                 // datos incompletos: no se registra nada
    }
    descripcionIrregularidad = descripcion;
    severidad = nuevaSeveridad;
    return true;
}

// [SOBRESCRITURA] Para una ficha, "iniciar el proceso" = evaluar la salud
// y dejar constancia en las observaciones.
void Ficha_Medica::iniciarProceso()
{
    estadoSaludAdecuado = evaluacionSalud();
    if (estadoSaludAdecuado)
    {
        agregarObservaciones("Evaluacion de salud: adecuada"); // metodo heredado del padre
    }
    else
    {
        agregarObservaciones("Evaluacion de salud: NO adecuada");
    }

    if (getIrregularidades()) {
        agregarObservaciones("Irregularidad (" + getSeveridadTexto() + "): " + descripcionIrregularidad);
    }
}

// [SOBRESCRITURA] Toma el reporte del padre y le agrega los datos de la ficha.
std::string Ficha_Medica::generarReporte() const
{
    std::ostringstream temp;
    temp << std::fixed << std::setprecision(1) << temperatura; // 36.6 en vez de 36.600000

    std::string resultado = Registro::generarReporte(); // 'Registro::' = version del padre
    resultado += " | [Ficha Medica] Golpes: ";
    resultado += (tieneGolpes ? "Si" : "No");
    resultado += " | Temp: " + temp.str();
    resultado += " | Irregularidades: ";
    if (getIrregularidades()) {
        resultado += "Si (" + getSeveridadTexto() + ": " + descripcionIrregularidad + ")";
    } else {
        resultado += "No";
    }
    resultado += " | Salud adecuada: ";
    resultado += (estadoSaludAdecuado ? "Si" : "No");
    return resultado;
}

#include "Guarderia.h"
#include "Validaciones.h"   // reglas de formato (ids y telefonos)

Guarderia::Guarderia(const std::string& nombreGuarderia, int capacidadMaxima)
    : nombreGuarderia(nombreGuarderia),
      capacidadMaxima(capacidadMaxima),
      cantidadActual(0)               // al inicio no hay infantes
{
}

// GUARD CLAUSES: primero se revisa todo lo que IMPEDIRIA la operacion y se sale
// temprano con 'return false'. El caso feliz queda al final, sin if anidados.
bool Guarderia::inscribirInfante(const std::shared_ptr<Infante>& infante) {
    if (!infante) {                   // puntero vacio (nullptr)
        return false;
    }
    if (cantidadActual >= capacidadMaxima) {   // sin cupo
        return false;
    }
    // RED DE SEGURIDAD: el menu ya valida lo que escribe el usuario, pero la
    // guarderia tambien rechaza datos con mal formato (asi nadie puede colarlos
    // creando infantes por otro camino).
    if (!esIdInfanteValido(infante->getIdInfante())) {
        return false;
    }
    for (const Tutor& t : infante->getTutores()) {
        if (!esIdTutorValido(t.getIdTutor()) || !esTelefonoValido(t.getTelefono())) {
            return false;
        }
    }
    if (buscarInfante(infante->getIdInfante()) != nullptr) {   // id repetido
        return false;
    }
    // Ya NO se copia el infante: la guarderia y quien lo creo COMPARTEN el mismo
    // objeto. (Copiar un Infante copiaria el puntero del expediente, no el
    // expediente: dos infantes "distintos" compartirian uno solo.)
    infantes.push_back(infante);
    // static_cast<int>: size() devuelve un tipo sin signo (size_t); convertimos
    // a int de forma explicita porque cantidadActual es int.
    cantidadActual = static_cast<int>(infantes.size());
    return true;
}

bool Guarderia::eliminarInfante(const std::string& idInfante) {
    // size_t es el tipo correcto para recorrer un vector por indice.
    for (std::size_t i = 0; i < infantes.size(); ++i) {
        if (infantes[i]->getIdInfante() == idInfante) {   // '->' accede a un miembro mediante un puntero
            // erase necesita una POSICION (iterador), no un numero:
            // begin() es la posicion del primero; sumarle i avanza hasta el elemento i.
            infantes.erase(infantes.begin() + i);
            cantidadActual = static_cast<int>(infantes.size());
            return true;
        }
    }
    return false;                     // recorrimos todo y no estaba
}

// [ACTUALIZACION]: Permite ejecutar la salida de emergencia de un infante en la guarderia,
// actualizando su estatus y removiendolo de la ocupacion actual.
bool Guarderia::registrarSalidaEmergencia(const std::string& idInfante) {
    std::shared_ptr<Infante> inf = buscarInfante(idInfante);
    if (!inf) {
        return false;
    }
    inf->actualizarEstatus("Salida de Emergencia");
    return eliminarInfante(idInfante);
}

std::shared_ptr<Infante> Guarderia::buscarInfante(const std::string& idInfante) const {
    for (const std::shared_ptr<Infante>& inf : infantes) {
        if (inf->getIdInfante() == idInfante) {
            return inf;               // devolvemos el puntero (no una copia del infante)
        }
    }
    return nullptr;                   // senal de "no encontrado"
}

// Junta los tutores de TODOS los infantes sin repetir (se compara por idTutor).
// Cada Infante guarda sus propios tutores, por eso hay que recorrerlos todos.
std::vector<Tutor> Guarderia::obtenerTutores() const {
    std::vector<Tutor> resultado;
    for (const std::shared_ptr<Infante>& inf : infantes) {
        for (const Tutor& t : inf->getTutores()) {
            bool yaEsta = false;
            for (const Tutor& agregado : resultado) {
                if (agregado.getIdTutor() == t.getIdTutor()) {
                    yaEsta = true;
                    break;                        // ya lo teniamos: dejamos de buscar
                }
            }
            if (!yaEsta) {
                resultado.push_back(t);
            }
        }
    }
    return resultado;
}

}

// Solo acepta los estatus validos. Devuelve false (y NO cambia nada) si no.
// [ACTUALIZACION]: Se agrega "Salida de Emergencia" a los estatus permitidos.
bool Infante::actualizarEstatus(const std::string& estatus) {
    // Los string distinguen mayusculas: "aceptado" != "Aceptado".
    bool esValido = (estatus == "Pendiente" ||
                     estatus == "Aceptado"  ||
                     estatus == "Rechazado" ||
                     estatus == "Salida de Emergencia");

const Tutor* Infante::buscarTutor(const std::string& idTutor) const {
    for (const Tutor& t : tutores) {
        if (t.getIdTutor() == idTutor) {
            return &t;
        }
    }
    return nullptr;

// Agrega un tutor ADICIONAL. Se rechaza si el formato es invalido o si ese id
// ya es tutor de este infante.
bool Infante::agregarTutor(const Tutor& tutor) {
    if (!esIdTutorValido(tutor.getIdTutor()) || !esTelefonoValido(tutor.getTelefono())) {
        return false;
    }
    if (tieneTutor(tutor.getIdTutor())) {
        return false;
    }
    tutores.push_back(tutor);
    return true;
}

// Elimina un tutor. La multiplicidad 1..* del UML exige que SIEMPRE quede al
// menos uno, asi que se rechaza si es el unico (para eso existe reemplazarTutor).
bool Infante::eliminarTutor(const std::string& idTutor) {
    if (tutores.size() <= 1) {
        return false;
    }
    for (std::size_t i = 0; i < tutores.size(); ++i) {
        if (tutores[i].getIdTutor() == idTutor) {
            tutores.erase(tutores.begin() + i);   // erase pide una POSICION, no un numero
            return true;
        }
    }
    return false;                                 // no estaba
}

// Reemplaza un tutor por otro, en la MISMA posicion de la lista.
// Como no cambia la cantidad de tutores, funciona tambien con el unico tutor.
bool Infante::reemplazarTutor(const std::string& idTutorActual, const Tutor& tutorNuevo) {
    if (!esIdTutorValido(tutorNuevo.getIdTutor()) || !esTelefonoValido(tutorNuevo.getTelefono())) {
        return false;
    }
    // El id nuevo no puede coincidir con OTRO tutor de este infante
    // (si puede ser el mismo id: seria actualizar los datos de ese tutor).
    if (tutorNuevo.getIdTutor() != idTutorActual && tieneTutor(tutorNuevo.getIdTutor())) {
        return false;
    }
    for (Tutor& t : tutores) {                    // 'Tutor&': el tutor REAL, no una copia
        if (t.getIdTutor() == idTutorActual) {
            t = tutorNuevo;                       // la asignacion reemplaza todo su contenido
            return true;
        }
    }
    return false;                                 // el tutor a reemplazar no existe
}

    return os;

// [SOBRESCRITURA] Aqui "iniciar el proceso" = abrir el tramite de ingreso.

    motivoRechazo = "";
    advertencia = "";
    fechaHoraEntrada = "";
    fechaHoraRechazo = "";
    infante->actualizarEstatus("Pendiente"); // modifica al infante REAL (compartido)
    agregarObservaciones("Proceso de ingreso iniciado");
}

// Decide el ingreso. Cada comprobacion que falla guarda el motivo, rechaza y sale.
bool Ingreso_Guarderia::validarAcceso(const Ficha_Medica &ficha)
{
    // 1) La ficha debe pertenecer a ESTE infante (la identidad se verifica antes que la salud).
    if (ficha.getInfante()->getIdInfante() != infante->getIdInfante())
    {
        motivoRechazo = "La ficha medica pertenece a otro infante";
        rechazarYActualizarEstatus();
        return false;
    }
    // 2) Una irregularidad GRAVE impide el ingreso. Se revisa ANTES que la salud
    //    general para poder dar un motivo preciso.
    if (ficha.getSeveridad() == SeveridadIrregularidad::Grave)
    {
        motivoRechazo = "Irregularidad grave: " + ficha.getDescripcionIrregularidad();
        rechazarYActualizarEstatus();
        return false;
    }
    // 3) El estado de salud (golpes y temperatura) debe ser adecuado.
    //    (evaluacionSalud() es const: solo consulta.)
    if (!ficha.evaluacionSalud())
    {
        motivoRechazo = "Estado de salud no adecuado";
        rechazarYActualizarEstatus();
        return false;
    }
    // 4) Todo en orden: se aprueba. Si la ficha trae una irregularidad de tipo
    //    Advertencia, se aprueba CON advertencia y queda constancia de ella.
    ingresoAprobado = true;
    motivoRechazo = "";
    fechaHoraEntrada = fechaHoraActual();   // se guarda la fecha y hora de entrada
    fechaHoraRechazo = "";
    infante->actualizarEstatus("Aceptado");
    if (ficha.getSeveridad() == SeveridadIrregularidad::Advertencia)
    {
        advertencia = ficha.getDescripcionIrregularidad();
        agregarObservaciones("Ingreso aprobado CON ADVERTENCIA: " + advertencia);
    }
    else
    {
        advertencia = "";
        agregarObservaciones("Ingreso aprobado");
    }
    return true;
}

// Unico punto de rechazo: asi no se repite este codigo en cada comprobacion.
void Ingreso_Guarderia::rechazarYActualizarEstatus()
{
    ingresoAprobado = false;
    advertencia = "";
    fechaHoraEntrada = "";
    fechaHoraRechazo = fechaHoraActual();   // se guarda la fecha y hora del rechazo
    infante->actualizarEstatus("Rechazado");
    agregarObservaciones("Ingreso rechazado");
}

// [SOBRESCRITURA] Reporte del padre + el resultado del ingreso.
std::string Ingreso_Guarderia::generarReporte() const
{
    std::string resultado = Registro::generarReporte();
    resultado += " | [Ingreso] Aprobado: ";
    resultado += (ingresoAprobado ? "Si" : "No");
    if (ingresoAprobado) {
        resultado += " | Entrada: " + fechaHoraEntrada;
    } else if (!fechaHoraRechazo.empty()) {
        resultado += " | Rechazado: " + fechaHoraRechazo;
    }
    if (!advertencia.empty()) {
        resultado += " (CON ADVERTENCIA: " + advertencia + ")";
    }
    if (!ingresoAprobado && !motivoRechazo.empty())
    {
        resultado += " | Motivo: " + motivoRechazo;
    }
    return resultado;
}

            std::cout << "Valor invalido. Intenta de nuevo.\n";
        }
    }
}

// Rango razonable de temperatura corporal (en grados C). Cambialo aqui si quieres otro.
const float TEMP_MINIMA = 30.0f;
const float TEMP_MAXIMA = 45.0f;

// Repite la pregunta hasta que la temperatura este dentro del rango.
float leerTemperatura(const std::string& mensaje) {
    while (true) {
        float t = leerDecimal(mensaje);
        if (!std::cin) {
            return 0;                          // la entrada se cerro: salimos del ciclo
        }
        if (t >= TEMP_MINIMA && t <= TEMP_MAXIMA) {
            return t;
        }
        std::cout << "  Temperatura fuera de rango (" << TEMP_MINIMA << " a "
                  << TEMP_MAXIMA << " C). Intenta de nuevo.\n";
    }
}

// Pregunta de si/no: devuelve true si la respuesta empieza con 's' o 'S'.
bool leerSiNo(const std::string& mensaje) {
    std::string texto = leerTexto(mensaje);
    return !texto.empty() && (texto[0] == 's' || texto[0] == 'S');
}

// ---- Lecturas CON VALIDACION -------------------------------------------
// Las tres siguen la misma idea: repiten la pregunta hasta que el dato tenga
// el formato correcto. Si el usuario deja la linea VACIA devuelven "" y quien
// las llamo interpreta eso como "cancelar" (asi nadie queda atrapado en un
// ciclo). Tambien devuelven "" si la entrada se cerro.

// Id de infante: I-XXXX.  aMayusculas() hace que "i-0001" tambien se acepte.
std::string leerIdInfante(const std::string& mensaje) {
    while (true) {
        std::string id = aMayusculas(leerTexto(mensaje));
        if (id.empty() || esIdInfanteValido(id)) {
            return id;
        }
        std::cout << "  Formato invalido. Debe ser " << formatoId('I')
                  << " (X = digito, ej. I-0001). Deja vacio para cancelar.\n";
    }
}

// Id de tutor: T-XXXX.
std::string leerIdTutor(const std::string& mensaje) {
    while (true) {
        std::string id = aMayusculas(leerTexto(mensaje));
        if (id.empty() || esIdTutorValido(id)) {
            return id;
        }
        std::cout << "  Formato invalido. Debe ser " << formatoId('T')
                  << " (X = digito, ej. T-0001). Deja vacio para cancelar.\n";
    }
}

// Telefono: exactamente DIGITOS_TELEFONO digitos, solo numeros.
std::string leerTelefono(const std::string& mensaje) {
    while (true) {
        std::string telefono = leerTexto(mensaje);
        if (telefono.empty() || esTelefonoValido(telefono)) {
            return telefono;
        }
        std::cout << "  Telefono invalido. Deben ser exactamente " << DIGITOS_TELEFONO
                  << " digitos, solo numeros. Deja vacio para cancelar.\n";
    }
}

// Gravedad de una irregularidad. Devuelve Ninguna si el usuario escribe 0 (= cancelar)
// o si la entrada se cerro; asi este ciclo nunca queda atrapado.
SeveridadIrregularidad leerSeveridad(const std::string& mensaje) {
    while (true) {
        int opcion = leerEntero(mensaje);
        if (opcion == 1) {
            return SeveridadIrregularidad::Advertencia;
        }
        if (opcion == 2) {
            return SeveridadIrregularidad::Grave;
        }
        if (opcion == 0) {
            return SeveridadIrregularidad::Ninguna;
        }
        std::cout << "  Opcion invalida. Escribe 1, 2 o 0.\n";
    }
}

// Captura los datos de un tutor pregunta por pregunta, validando cada uno.
// Devuelve false si el usuario cancela (deja un dato vacio); si no, deja el
// tutor en 'destino' (se recibe por REFERENCIA para poder llenarlo).
bool capturarTutor(Tutor& destino) {
    std::string idTutor = leerIdTutor("  Id del tutor (formato " + formatoId('T') + "): ");
    if (idTutor.empty()) {
        return false;
    }
    std::string nombre = leerTexto("  Nombre: ");
    std::string telefono = leerTelefono("  Telefono (" + std::to_string(DIGITOS_TELEFONO) + " digitos): ");
    if (nombre.empty() || telefono.empty()) {
        return false;
    }
    std::string parentesco = leerTexto("  Parentesco: ");
    destino = Tutor(idTutor, nombre, telefono, parentesco);
    return true;
}

// Fecha de hoy con formato "AAAA-MM-DD" (para los registros).
std::string fechaHoy() {
    std::time_t ahora = std::time(nullptr);          // segundos desde 1970
    std::tm* local = std::localtime(&ahora);         // los convierte a fecha local
    char buffer[11];                                 // "AAAA-MM-DD" + terminador = 11
    std::strftime(buffer, sizeof(buffer), "%Y-%m-%d", local);
    return buffer;
}

// OPCION 1: pide los datos, arma las piezas (Tutor, Expediente, Infante) y las inscribe.
void MenuGuarderia::opcionInscribir() {
    if (!guarderia.hayCupo()) {
        std::cout << "La guarderia esta llena (" << guarderia.contarInfantes() << " de "
                  << guarderia.getCapacidadMaxima() << "). Elimina un infante (opcion 7) para liberar un cupo.\n";
        return;
    }
    std::cout << "Cupos libres: " << guarderia.getCapacidadMaxima() - guarderia.contarInfantes()
              << " de " << guarderia.getCapacidadMaxima() << ". (Deja un dato vacio para cancelar.)\n";

    std::string id = leerIdInfante("Id del infante (formato " + formatoId('I') + "): ");
    if (id.empty()) {
        std::cout << "Inscripcion cancelada.\n";
        return;
    }
    if (guarderia.buscarInfante(id) != nullptr) {    // nullptr = no existe; si NO es nullptr, ya existe
        std::cout << "Ya existe un infante con ese id.\n";
        return;
    }
    std::string nombre = leerTexto("Nombre completo: ");
    if (nombre.empty()) {
        std::cout << "Inscripcion cancelada.\n";
        return;
    }

    // El tutor se puede capturar de DOS formas:
    //   a) en UNA linea con el formato id|nombre|telefono|parentesco, usando el
    //      [OPERADOR >>] sobrecargado de Tutor sobre cin;
    //   b) pregunta por pregunta (cada dato se valida al momento).
    Tutor tutor("", "", "", "");                     // tutor "en blanco" que se llenara
    if (leerSiNo("Capturar el tutor en UNA linea (id|nombre|telefono|parentesco)? (s/n): ")) {
        std::cout << "Tutor: ";
        // 'cin >> tutor' llama a operator>>(istream&, Tutor&). Devuelve el flujo, y un
        // flujo se puede usar como condicion: es false si la lectura FALLO.
        if (!(std::cin >> tutor)) {
            std::cin.clear();                        // limpia el estado de error del flujo
            std::cout << "Formato incorrecto. No se inscribio.\n";
            return;
        }
        // operator>> solo comprueba que haya 4 campos. Las reglas de formato las
        // comprobamos aqui, igual que en la captura pregunta por pregunta.
        tutor = Tutor(aMayusculas(tutor.getIdTutor()), tutor.getNombre(),
                      tutor.getTelefono(), tutor.getParentesco());
        if (!esIdTutorValido(tutor.getIdTutor())) {
            std::cout << "Id de tutor invalido (debe ser " << formatoId('T')
                      << "). No se inscribio.\n";
            return;
        }
        if (!esTelefonoValido(tutor.getTelefono())) {
            std::cout << "Telefono invalido (deben ser exactamente " << DIGITOS_TELEFONO
                      << " digitos, solo numeros). No se inscribio.\n";
            return;
        }
    } else {
        // IMPORTANTE: cada dato se lee en su PROPIA variable. Si escribieramos
        // Tutor(leerTexto(..), leerTexto(..), ...), C++ NO garantiza el orden en que
        // se evaluan los argumentos de una funcion: las preguntas podrian salir
        // desordenadas.
        std::string idTutor = leerIdTutor("  Id del tutor (formato " + formatoId('T') + "): ");
        if (idTutor.empty()) {
            std::cout << "Inscripcion cancelada.\n";
            return;
        }
        std::string nombreTutor = leerTexto("  Nombre: ");
        std::string telefono = leerTelefono("  Telefono (" + std::to_string(DIGITOS_TELEFONO) + " digitos): ");
        if (nombreTutor.empty() || telefono.empty()) {
            std::cout << "Inscripcion cancelada.\n";
            return;
        }
        std::string parentesco = leerTexto("  Parentesco: ");
        tutor = Tutor(idTutor, nombreTutor, telefono, parentesco);   // reemplaza el tutor en blanco
    }

    int tipo = leerEntero("Tipo de expediente (1 = Lactante, 2 = Preescolar): ");
    std::string idExpediente = "EXP-" + id;

    // Puntero al PADRE: apuntara a una hija u otra segun lo que elija el usuario.
    std::shared_ptr<Expediente_Medico> expediente;

    if (tipo == 1) {
        int meses = leerEntero("Edad en meses: ");
        std::string alimentacion = leerTexto("Alimentacion (materna/formula/mixta): ");
        float peso = leerDecimal("Peso en kg: ");
        expediente = std::make_shared<Expediente_Lactante>(idExpediente, meses, alimentacion, peso);
    } else if (tipo == 2) {
        int anios = leerEntero("Edad en anios: ");
        bool esfinteres = leerSiNo("Controla esfinteres? (s/n): ");
        std::string sala = leerTexto("Sala asignada: ");
        expediente = std::make_shared<Expediente_Preescolar>(idExpediente, anios, esfinteres, sala);
    } else {
        std::cout << "Tipo invalido. No se inscribio.\n";
        return;
    }

    // Alergias: el usuario decide si agrega y si indica severidad.
    while (leerSiNo("Agregar una alergia? (s/n): ")) {
        std::string alergia = leerTexto("  Alergia: ");
        if (leerSiNo("  Indicar severidad? (s/n): ")) {
            std::string severidad = leerTexto("  Severidad: ");
            expediente->agregarAlergia(alergia, severidad);   // [SOBRECARGA] version de 2 parametros
        } else {
            expediente->agregarAlergia(alergia);              // [SOBRECARGA] version de 1 parametro
        }
    }

    // make_shared crea el Infante y su puntero inteligente de una vez.
    std::shared_ptr<Infante> infante = std::make_shared<Infante>(id, nombre, tutor, expediente);
    if (guarderia.inscribirInfante(infante)) {
        std::cout << "Infante inscrito (" << expediente->getCategoria() << "). Cupos libres: "
                  << guarderia.getCapacidadMaxima() - guarderia.contarInfantes() << ".\n";
    } else {
        std::cout << "No se pudo inscribir.\n";
    }
}

    std::cout << "   Formato de texto (operador <<):\n" << *inf;
}

// OPCION 5: el PROCESO DE INGRESO. Usa herencia, polimorfismo y la dependencia
// entre Ingreso_Guarderia y Ficha_Medica.
void MenuGuarderia::opcionIngreso() {
    std::string id = aMayusculas(leerTexto("Id del infante: "));
    std::shared_ptr<Infante> inf = guarderia.buscarInfante(id);
    if (!inf) {
        std::cout << "No se encontro ese infante.\n";
        return;
    }
    bool golpes = leerSiNo("Presenta golpes? (s/n): ");
    float temperatura = leerTemperatura("Temperatura en C: ");

    // IRREGULARIDADES: se pregunta si presenta alguna y, si es asi, cual y que tan seria.
    // Se captura ANTES de crear los registros: si el usuario cancela, no se crea nada
    // (asi nunca se aprueba por error a un infante con una irregularidad sin registrar).
    SeveridadIrregularidad severidad = SeveridadIrregularidad::Ninguna;
    std::string descripcion;
    if (leerSiNo("Presenta alguna irregularidad? (s/n): ")) {
        descripcion = leerTexto("  Cual es la irregularidad?: ");
        if (descripcion.empty()) {
            std::cout << "Proceso cancelado.\n";
            return;
        }
        severidad = leerSeveridad("  Gravedad (1 = Advertencia: pasa con advertencia, 2 = Grave: no pasa, 0 = cancelar): ");
        if (severidad == SeveridadIrregularidad::Ninguna) {
            std::cout << "Proceso cancelado.\n";
            return;
        }
    }

    ++contadorRegistros;                             // numero consecutivo para los ids de los registros
    std::string fecha = fechaHoy();
    // Ambos registros apuntan al MISMO infante de la guarderia.
    std::shared_ptr<Ficha_Medica> ficha = std::make_shared<Ficha_Medica>(
        "F-" + std::to_string(contadorRegistros), fecha, inf, golpes, temperatura);
    std::shared_ptr<Ingreso_Guarderia> ingreso = std::make_shared<Ingreso_Guarderia>(
        "G-" + std::to_string(contadorRegistros), fecha, inf);

    if (severidad != SeveridadIrregularidad::Ninguna) {
        ficha->registrarIrregularidad(descripcion, severidad);
    }

    // [POLIMORFISMO] Un vector de punteros al PADRE (Registro) que contiene
    // objetos de las HIJAS. Un shared_ptr<Hija> se convierte solo en shared_ptr<Padre>.
    std::vector<std::shared_ptr<Registro>> proceso;
    proceso.push_back(ficha);
    proceso.push_back(ingreso);

    // Misma llamada, comportamiento distinto: la ficha evalua la salud y el
    // ingreso abre el tramite. C++ elige la version segun el objeto REAL.
    for (const std::shared_ptr<Registro>& r : proceso) {
        r->iniciarProceso();
    }

    // Dependencia: el ingreso usa la ficha como parametro para decidir.
    // '*ficha' convierte el puntero en el objeto Ficha_Medica.
    bool aprobado = ingreso->validarAcceso(*ficha);
    std::cout << (aprobado ? "INGRESO APROBADO" : "INGRESO RECHAZADO");
    if (aprobado && !ingreso->getAdvertencia().empty()) {
        std::cout << " CON ADVERTENCIA - " << ingreso->getAdvertencia();
    }
    if (!aprobado) {
        std::cout << " - " << ingreso->getMotivoRechazo();
    }
    std::cout << "\n";

    // generarReporte() tambien es virtual: cada hija agrega lo suyo.
    for (const std::shared_ptr<Registro>& r : proceso) {
        historial.push_back(r);                      // se conserva para la opcion 6
        std::cout << r->generarReporte() << "\n";
    }
}

// OPCION 6: imprime el historial de todos los registros del programa.
void MenuGuarderia::opcionHistorial() const {
    if (historial.empty()) {
        std::cout << "Aun no hay registros.\n";
        return;
    }
    for (const std::shared_ptr<Registro>& r : historial) {
        std::cout << r->generarReporte() << "\n";    // polimorfismo otra vez
    }
}

// OPCION 7: elimina un infante (pidiendo confirmacion, porque no se puede deshacer).
// Nota: los registros que ya esten en el historial se CONSERVAN (el historial
// guarda su propio puntero compartido al infante), igual que en una guarderia
// real el historial no se borra cuando un nino se da de baja.
void MenuGuarderia::opcionEliminar() {
    if (guarderia.contarInfantes() == 0) {
        std::cout << "No hay infantes inscritos.\n";
        return;
    }
    std::string id = aMayusculas(leerTexto("Id del infante a eliminar (formato " + formatoId('I') + "): "));
    std::shared_ptr<Infante> inf = guarderia.buscarInfante(id);
    if (!inf) {
        std::cout << "No se encontro ese infante.\n";
        return;
    }
    mostrarInfante(*inf);                            // se muestra a quien se va a eliminar
    if (!leerSiNo("Seguro que quieres eliminarlo? No se puede deshacer (s/n): ")) {
        std::cout << "No se elimino nada.\n";
        return;
    }
    if (guarderia.eliminarInfante(id)) {             // devuelve true si lo encontro y lo borro
        std::cout << "Infante eliminado. Cupos libres: "
                  << guarderia.getCapacidadMaxima() - guarderia.contarInfantes() << ".\n";
    } else {
        std::cout << "No se pudo eliminar.\n";
    }
}

// ---------------------------------------------------------------------
//  OPCION 9: ADMINISTRAR TUTORES (submenu)

// 9 -> 1: lista cada tutor UNA vez y debajo los infantes de los que es tutor.
void MenuGuarderia::opcionListarTutores() const {
    std::vector<Tutor> tutores = guarderia.obtenerTutores();
    if (tutores.empty()) {
        std::cout << "No hay tutores registrados.\n";
        return;
    }
    for (const Tutor& t : tutores) {
        std::cout << t.getIdTutor() << " | " << t.getNombre() << " | Tel: " << t.getTelefono() << "\n";
        std::cout << "   Tutor de:\n";

// Funciona con cualquier std::ostream (cout, un flujo de texto en memoria, etc.).

std::ostream& operator<<(std::ostream& os, const Tutor& tutor) {

       << tutor.getTelefono() << '|' << tutor.getParentesco();
    return os;

// [OPERADOR >>] Lee UNA linea con el mismo formato que escribe operator<<.
// Si el formato es incorrecto, marca el flujo como fallido: asi quien
// lo usa puede preguntar "if (flujo >> tutor)" y saber si la lectura funciono.
std::istream& operator>>(std::istream& is, Tutor& tutor) {
    std::string linea;
    if (std::getline(is, linea)) {                         // lee la linea completa
        std::vector<std::string> campos = dividir(linea, '|');
        if (campos.size() >= 4) {
            tutor = Tutor(campos[0], campos[1], campos[2], campos[3]);   // reemplaza el contenido
        } else {
            is.setstate(std::ios::failbit);                // faltan campos: lectura fallida
        }
    }
    return is;
}

#pragma once
#include <string>
#include <vector>
#include <sstream>   // std::istringstream: leer desde un texto como si fuera un flujo
#include <ctime>     // fecha y hora actual

// dividir("a|b|c", '|')  ->  ["a", "b", "c"]
inline std::vector<std::string> dividir(const std::string& texto, char separador) {
    std::vector<std::string> partes;      // aqui acumulamos los pedazos
    std::string parte;                    // el pedazo que estamos leyendo
    std::istringstream flujo(texto);      // convierte el texto en un flujo de lectura

    // getline con 3 parametros lee hasta encontrar el separador (en vez de leer hasta el salto de linea que es lo normal).
    while (std::getline(flujo, parte, separador)) {
        partes.push_back(parte);
    }

    // getline NO devuelve el campo vacio final de "a|b|" sino que se lo "traga" por asi decirlo.
    // Lo agregamos a mano para no perder el ultimo campo cuando esta vacio
    // (por ejemplo, un tutor al que le falta el ultimo dato).
    if (!texto.empty() && texto.back() == separador) {
        partes.push_back("");
    }
    return partes;
}

// Fecha y hora actual con formato "AAAA-MM-DD HH:MM:SS" (para la hora de entrada).
inline std::string fechaHoraActual() {
    std::time_t ahora = std::time(nullptr);          // segundos desde 1970
    std::tm* local = std::localtime(&ahora);         // los convierte a fecha y hora local
    char buffer[20];                                 // "AAAA-MM-DD HH:MM:SS" = 19 + terminador = 20
    std::strftime(buffer, sizeof(buffer), "%Y-%m-%d %H:%M:%S", local);
    return buffer;
}

// unir(["a","b","c"], ';')  ->  "a;b;c"
inline std::string unir(const std::vector<std::string>& partes, char separador) {
    std::string resultado;
    for (std::size_t i = 0; i < partes.size(); ++i) {
        if (i > 0) {                      // el separador va ENTRE elementos, no al inicio
            resultado += separador;
        }
        resultado += partes[i];
    }
    return resultado;
}

===== ARCHIVO: Validaciones.h =====
#pragma once
#include <string>
#include <cctype>      // std::toupper

// --- Reglas ---
const int DIGITOS_ID = 4;          // I-0001 y T-0001 llevan 4 digitos
const int DIGITOS_TELEFONO = 10;   // telefono de 10 digitos

// true si el texto NO esta vacio y TODOS sus caracteres son digitos '0'..'9'.
inline bool soloDigitos(const std::string& texto) { // inline: es una palabra clave que se coloca antes de una funcion
                                                   // para indicarle al compilador que inserte el cuerpo de la funcion directamente en cada lugar donde sea llamada, 
                                                   // en vez de hacer una llamada a funcion tradicional.
    if (texto.empty()) {
        return false;
    }
    for (char c : texto) {
        if (c < '0' || c > '9') {          // los caracteres son numeros: '0'..'9' van seguidos
            return false;
        }
    }
    return true;
}

// Comprueba el formato "<letra>-<N digitos>". Ejemplo: ("I-0001", 'I', 4) -> true
inline bool tieneFormatoId(const std::string& texto, char letra, int digitos) {
    // Largo esperado: letra + guion + los digitos.
    if (texto.size() != static_cast<std::size_t>(2 + digitos)) {
        return false;
        // static_cast es el operador oficial de C++ para realizar conversiones explicitas de tipo de datos en tiempo de compilacion de forma segura y controlada.
        // Convierte el int (con signo) al tipo sin signo que devuelve size().
        // Comparar ambos tipos sin convertir provoca una advertencia del compilador 
        // y puede dar resultados no esperados con numeros negativos
    }
    if (texto[0] != letra || texto[1] != '-') {
        return false;
    }
    return soloDigitos(texto.substr(2));   // substr(2): todo lo que sigue a "I-"
}

inline bool esIdInfanteValido(const std::string& id) {
    return tieneFormatoId(id, 'I', DIGITOS_ID);
}

inline bool esIdTutorValido(const std::string& id) {
    return tieneFormatoId(id, 'T', DIGITOS_ID);
}

inline bool esTelefonoValido(const std::string& telefono) {
    return telefono.size() == static_cast<std::size_t>(DIGITOS_TELEFONO) && soloDigitos(telefono);
}

// Texto de ayuda para mostrar el formato esperado: "I-XXXX" o "T-XXXX".
inline std::string formatoId(char letra) {
    return std::string(1, letra) + "-" + std::string(DIGITOS_ID, 'X');
}

// ---- Ayudas para limpiar lo que escribe el usuario --------------------

// Convierte a MAYUSCULAS (asi "i-0001" tambien se acepta como "I-0001").
// Recibe el texto POR VALOR (una copia) para poder modificarlo y devolverlo.
inline std::string aMayusculas(std::string texto) {
    for (char& c : texto) {                // 'char&': modifica cada caracter del texto original
        // toupper pide un unsigned char; el doble cast evita problemas con caracteres raros.
        c = static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
    }
    return texto;
}

// Quita los espacios (y tabuladores) del inicio y del final: "  I-0001 " -> "I-0001".
inline std::string recortar(const std::string& texto) {
    const std::string espacios = " \t\r\n";
    std::size_t inicio = texto.find_first_not_of(espacios);   // primer caracter que NO es espacio
    if (inicio == std::string::npos) {
        return "";                                            // era todo espacios (o vacio)
    }
    std::size_t fin = texto.find_last_not_of(espacios);       // ultimo caracter que NO es espacio
    return texto.substr(inicio, fin - inicio + 1);
}
